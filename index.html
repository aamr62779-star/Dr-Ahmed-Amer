from flask import Flask, request, redirect, url_for, render_template_string import os import sqlite3 from datetime import datetime app = Flask(__name__) app.secret_key = "dr_amer_clinic_final_secure_key" # مسارات حفظ الملفات وقاعدة البيانات لعيادة الدكتور أحمد UPLOAD_FOLDER = 'static/uploads/xrays' app.config['UPLOAD_FOLDER'] = UPLOAD_FOLDER DB_NAME = 'clinic.db' # --------------------------------------------------------- # تهيئة قاعدة البيانات وإنشاء الجداول المبسطة (بدون خريطة أسنان) # --------------------------------------------------------- def get_db_connection(): conn = sqlite3.connect(DB_NAME) conn.row_factory = sqlite3.Row return conn def init_db(): conn = get_db_connection() # جدول المرضى المبسط والخالي من تعقيد خريطة الأسنان conn.execute(''' CREATE TABLE IF NOT EXISTS patients ( id INTEGER PRIMARY KEY AUTOINCREMENT, patient_name TEXT, age_category TEXT, service TEXT, paid REAL, remaining REAL, date_added TEXT ) ''') # جدول المصاريف المؤمّن بالكامل conn.execute(''' CREATE TABLE IF NOT EXISTS expenses ( id INTEGER PRIMARY KEY AUTOINCREMENT, expense_title TEXT, expense_amount REAL, date_added TEXT ) ''') conn.commit() conn.close() # --------------------------------------------------------- # الواجهة الأمامية (HTML) مدمجة بالكامل وبدون خريطة الأسنان التفاعلية # --------------------------------------------------------- HTML_TEMPLATE = ''' Dr. Amer’s Clinic

Dr. Amer’s Clinic - إدارة العيادة الذكية

إيرادات الشهر الإجمالية

{{ "%.2f"|format(total_revenue) }} ج.م

مصروفات العيادة الكلية (هذا الشهر)

{{ "%.2f"|format(total_expenses) }} ج.م

صافي الربح الشهري

{{ "%.2f"|format(net_monthly_profit) }} ج.م

تسجيل مريض جديد

إضافة بند مصروفات

سجل المرضى

{% if patients %} {% for p in patients %} {% endfor %} {% else %} end {% endif %}
الاسم الفئة الإجراء الطبي المدفوع المتبقي التاريخ
{{ p.patient_name }} {{ p.age_category }} {{ p.service }} {{ p.paid }} ج.م {{ p.remaining }} ج.م {{ p.date_added.split(' ')[0] if p.date_added else '' }}
لا يوجد مرضى مسجلين حتى الآن.

بيان المصروفات التفصيلي لهذا الشهر

{% if expenses %} {% for e in expenses %} {% endfor %} {% else %} {% endif %}
بند المصروف المبلغ التاريخ والوقت
{{ e.expense_title }} {{ e.expense_amount }} ج.م {{ e.date_added }}
لم يتم تسجيل أي مصروفات حتى الآن.
''' # --------------------------------------------------------- # 1. الراوت الرئيسي: حساب الإحصائيات الشهرية وعرض الواجهة # --------------------------------------------------------- @app.route('/') def index(): try: conn = get_db_connection() current_month = datetime.now().strftime('%Y-%m') # جلب البيانات لملء الجداول patients = conn.execute('SELECT * FROM patients ORDER BY id DESC').fetchall() expenses = conn.execute('SELECT * FROM expenses ORDER BY id DESC').fetchall() # حساب الحسبة الشهرية الإجمالية بدقة لضمان العمولات total_revenue = conn.execute( "SELECT SUM(paid) FROM patients WHERE date_added LIKE ?", (f"{current_month}%",) ).fetchone()[0] or 0.0 total_expenses = conn.execute( "SELECT SUM(expense_amount) FROM expenses WHERE date_added LIKE ?", (f"{current_month}%",) ).fetchone()[0] or 0.0 net_monthly_profit = total_revenue - total_expenses conn.close() return render_template_string(HTML_TEMPLATE, patients=patients, expenses=expenses, total_revenue=total_revenue, total_expenses=total_expenses, net_monthly_profit=net_monthly_profit) except Exception as e: return f"🚨 تنبيه أمان: هناك مشكلة في السيرفر المحلي: {e}" # --------------------------------------------------------- # 2. دالة حفظ ملف المريض الآمنة # --------------------------------------------------------- @app.route('/save_patient', methods=['POST']) def save_patient(): try: patient_name = request.form.get('patient_name', '').strip() age_category = request.form.get('age_category', 'بالغ') service = request.form.get('service', '').strip() # حماية حسابات المريض من قيم النصوص غير الصالحة أو الفراغات paid_raw = request.form.get('paid', '0').strip() remaining_raw = request.form.get('remaining', '0').strip() try: paid = float(paid_raw) if paid_raw else 0.0 except ValueError: paid = 0.0 try: remaining = float(remaining_raw) if remaining_raw else 0.0 except ValueError: remaining = 0.0 if not patient_name: return redirect(url_for('index')) # حفظ صور الأشعة المتعددة بأمان xray_files = request.files.getlist('xray_images') for file in xray_files: if file and file.filename != '': if not os.path.exists(app.config['UPLOAD_FOLDER']): os.makedirs(app.config['UPLOAD_FOLDER']) filename = f"{patient_name}_{datetime.now().strftime('%Y%m%d%H%M%S')}_{file.filename}" file.save(os.path.join(app.config['UPLOAD_FOLDER'], filename)) today_date = datetime.now().strftime('%Y-%m-%d %H:%M:%S') conn = get_db_connection() conn.execute(''' INSERT INTO patients (patient_name, age_category, service, paid, remaining, date_added) VALUES (?, ?, ?, ?, ?, ?) ''', (patient_name, age_category, service, paid, remaining, today_date)) conn.commit() conn.close() return redirect(url_for('index')) except Exception as e: print(f"خطأ تم احتواؤه في المريض: {e}") return redirect(url_for('index')) # --------------------------------------------------------- # 3. دالة إضافة المصاريف [تم الإصلاح والـ Commit الفوري بنجاح] # --------------------------------------------------------- @app.route('/add_expense', methods=['POST']) def add_expense(): try: expense_title = request.form.get('expense_title', '').strip() expense_amount_raw = request.form.get('expense_amount', '0').strip() # طوق نجاة صارم لمنع سقوط السيرفر وتحويل الفراغات لـ صفر try: expense_amount = float(expense_amount_raw) if expense_amount_raw else 0.0 except ValueError: expense_amount = 0.0 if not expense_title or expense_amount <= 0: return redirect(url_for('index')) today_date = datetime.now().strftime('%Y-%m-%d %H:%M:%S') # الاتصال بالحفظ الفوري والتثبيت لضمان التسجيل بدون تهنيج conn = get_db_connection() conn.execute(''' INSERT INTO expenses (expense_title, expense_amount, date_added) VALUES (?, ?, ?) ''', (expense_title, expense_amount, today_date)) conn.commit() # تثبيت الحفظ فوراً في الداتابيز لضمان تسجيله conn.close() return redirect(url_for('index')) except Exception as e: print(f"خطأ تم منعه واحتواؤه في المصاريف: {e}") return redirect(url_for('index')) # --------------------------------------------------------- # التشغيل التلقائي الآمن للبرنامج والداتابيز # --------------------------------------------------------- if __name__ == '__main__': init_db() if not os.path.exists(UPLOAD_FOLDER): os.makedirs(UPLOAD_FOLDER) # تشغيل السيرفر المحلي للعيادة app.run(debug=True, port=5000)
